CREATE PROCEDURE "@AIS_BillDelivery_LoadPreSalesOrder" (IN PlanDate varchar(5000), IN BPGrp varchar(5000)) AS BEGIN SELECT T0."DocEntry", T0."DocNum", T0."U_CardCode", T0."U_CardName", T1."U_ItemCode", "U_ItemName", T3."U_ItemCategory", T1."U_DefUnit", SUM(IFNULL(T2."U_Qty", 0.0)) AS "BagQuantity", SUM(IFNULL(T2."U_Qty", 0.0)) AS "TonQuantity", T1."U_UnitPrice", T2."U_WhsCode", T2."U_WhsName", T1."U_TaxCode" FROM "@AIS_OPRE" T0 INNER JOIN "@AIS_PRE1" T1 ON T1."DocEntry" = T0."DocEntry" INNER JOIN "@AIS_PRE2" T2 ON T2."DocEntry" = T1."DocEntry" AND T2."U_UniqID" = T1."LineId" INNER JOIN OITM T3 ON T3."ItemCode" = T1."U_ItemCode" INNER JOIN OCRD T4 ON T4."CardCode" = T0."U_CardCode" WHERE CAST(T2."U_PlanDate" AS varchar(10)) = CAST(:PlanDate AS varchar(10)) AND (T4."GroupCode" = :BPGrp OR :BPGrp = '-1') AND IFNULL(T1."U_ItemCode", '') <> '' AND IFNULL(T2."U_WhsCode", '') <> '' AND IFNULL(T2."U_ReApprove", 'N') = 'Y' AND IFNULL(T2."U_Qty", 0.0) > 0.0 AND CAST(T0."DocEntry" AS varchar(5000)) || '_' || T1."U_ItemCode" || '_' || CAST(T2."U_PlanDate" AS varchar(10)) NOT IN (SELECT A1."U_PreSaleEnt" || '_' || A1."U_ItemCode" || '_' || CAST(A0."U_BillRecDate" AS varchar(10)) FROM "@AIS_LOAD" A0 INNER JOIN "@AIS_LOAD1" A1 ON A1."DocEntry" = A0."DocEntry" WHERE A1."U_PreSaleEnt" = T0."DocEntry" AND IFNULL(A1."U_Select", 'N') = 'Y') AND IFNULL(T1."U_MobStatus", 'N') <> 'D' AND IFNULL(t2."U_CompStatus", '-1') = '-1' AND IFNULL(t2."U_MobStatus", '-1') = '-1' GROUP BY T0."DocEntry", T0."DocNum", T0."U_CardCode", T0."U_CardName", T1."U_ItemCode", "U_ItemName", T2."U_WhsCode", T2."U_WhsName", T1."U_TaxCode", "U_UnitPrice", T3."U_ItemCategory", T1."U_DefUnit";
--[Note:Modifier] SAP HANA does not support function CONVERT; use the CAST function or implicit casting functions
END;
