R."CardCode" AS "NES_CardCode",R."CardName" AS "NES_CardName",R."CntctPrsn" as "NE_ContactPerson",
R."Cellular" AS "NE_Cellular",

(SELECT "GroupName" FROM "OCRG" WHERE "GroupCode"=R."GroupCode") as "NE_GroupName",
--R."U_AVA_CUS_TYPE" as "CustomerType",
(Select IFNULL(SUM(A."Debit"-A."Credit"),0) as "opening" from "JDT1" A where A."RefDate"<:FromD and A."ShortName"=R."CardCode")as "NE_Opening",
(select IFNULL(SUM(A."Debit"),0) from "JDT1" A where A."RefDate">=:FromD and A."RefDate"<=:ToD and A."ShortName"=R."CardCode") as "NE_Debit",
(select IFNULL(SUM(A."Credit"),0) from "JDT1" A where A."RefDate">=:FromD and A."RefDate"<=:ToD and A."ShortName"=R."CardCode") as "NE_Credit" ,
(Select IFNULL(SUM(A."Debit"-A."Credit"),0)as "closing balance" from JDT1 A where A."RefDate"<=:Tod and A."ShortName"=R."CardCode")as "NE_Closing",
R."E_Mail" as "E_Email-ID",:CCMail AS "E_CC Mail-ID",:Status as "E_Status" from JDT1 B

B."Debit" ,B."Credit",R."E_Mail")Z
Where IFNULL(Z."NE_Closing",0) > 0
Order by Z."NES_CardCode";
