CREATE LOCAL TEMPORARY TABLE "#Temp" (
DocEnt CLOB,
CardCode VARCHAR(100),
EmailID VARCHAR(254),
CCMailID VARCHAR(254),

);

"Name"='[%1]' and "U_AV_CrdCd"=A."U_AV_CrdCd" and "U_AV_CntPn"=A."U_AV_CntPn" 
) AS "DocEnt",                                        
IFNULL("U_AV_CrdCd",'') AS "CardCode",IFNULL("U_AV_EmlId",'') AS "EmailID",IFNULL("U_AV_CCmlId",'') AS "CCMailID",IFNULL("U_AV_CntPn",'') AS "ContactPerson" from "@AV_GRPTMP" A  
Where "Name"='[%1]' into "#Temp";

END
