DO BEGIN

  DECLARE FCC VARCHAR(100) := '%1';
  DECLARE TCC VARCHAR(100) := '%2';
  DECLARE FromF DATE := '%3';
  DECLARE To DATE := '%4';
  DECLARE BPGroup VARCHAR(20) := '%5';
  DECLARE Status VARCHAR(8000) := '';
  DECLARE SelectS VARCHAR(2) := '';
  DECLARE CCMail VARCHAR(254) := '%6';
  
  Select Row_Number() Over(Order by A."DocNum") AS "NES_#",:SelectS AS "CK_Select",A."DocEntry" AS "NE_DocEntry",A."DocNum" AS "NES_DocNum",A."CardCode" AS "NES_CardCode",A."CardName" AS "NES_CardName",B."CntctPrsn" AS "NE_ContactPerson",
  (SELECT "GroupName" FROM	"OCRG" WHERE "GroupCode"=B."GroupCode") AS "NE_GroupName",A."DocDate" AS "NES_Posting Date",A."DocDueDate" AS "NE_Due Date",
  A."DocTotal" AS "NE_DocTotal",(Select "E_MailL" from "OCPR" where "CardCode"=A."CardCode" and "Name"=B."CntctPrsn") AS "E_Email-ID",:CCMail AS "E_CC Mail-ID",
  :Status AS "E_Status" FROM "OINV" A INNER JOIN "OCRD" B ON A."CardCode"=B."CardCode" 
  WHERE (:FromF='' or A."DocDate">=:FromF) and (:To='' or A."DocDate"<=:To) 
  and (:FCC='' or A."CardCode">=:FCC) and (:TCC='' or A."CardCode"<=:TCC)
  and (:BPGroup='' or B."GroupCode"=(SELECT "GroupCode" FROM "OCRG" WHERE "GroupName"=:BPGroup))
  ORDER BY A."DocNum";

END;
